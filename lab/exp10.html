C Program - Student Information

#include <stdio.h> struct Student { char name[50]; char regno[20]; char address[100]; char phone[15]; int age; char course[50]; float attendance; float percentage; }; int main() { struct Student s; printf("Enter Student Name: "); scanf(" %[^\n]", s.name); printf("Enter Register Number: "); scanf("%s", s.regno); printf("Enter Address: "); scanf(" %[^\n]", s.address); printf("Enter Phone Number: "); scanf("%s", s.phone); printf("Enter Age: "); scanf("%d", &s.age); printf("Enter Course: "); scanf(" %[^\n]", s.course); printf("Enter Attendance Percentage: "); scanf("%f", &s.attendance); printf("Enter Mark Percentage: "); scanf("%f", &s.percentage); printf("\n----- STUDENT INFORMATION -----\n"); printf("Name : %s\n", s.name); printf("Register No: %s\n", s.regno); printf("Address : %s\n", s.address); printf("Phone : %s\n", s.phone); printf("Age : %d\n", s.age); printf("Course : %s\n", s.course); printf("Attendance : %.2f%%\n", s.attendance); printf("Percentage : %.2f%%\n", s.percentage); return 0; }
